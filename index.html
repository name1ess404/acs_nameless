#name
while True:
    name = input("Enter Your Name: ").strip()
    if name == "":  print("You didn't enter anything!"); continue
    elif any(char.isdigit() for char in name): print("Name can't contain numbers!"); continue
    else: break
    
#age
while True:
    try:
        age = int(input("Enter Your Age: "))
        if age <= 0:
            print("You aren't born yet."); continue
        elif age >= 120:
            print("You are already dead."); continue
        else: break

    except ValueError:
        print("Input needs to be an number!"); continue

#class
while True:
    try:
        class_ = int(input("Enter Your Class: "))
        if not(1 <= class_ <= 12):
            print("That isn't a class."); continue
        else: break
    except ValueError:
        print("Input needs to be an number!"); continue

#printing nicely
width = max(len(name), len(str(age)), len(str(class_)))
print() #new line

print("-" * ((width+10) + 3 + 6)) #top bar

print(f"|    Name  :  {name:<{width}}    |")
print(f"|    Age   :  {age:<{width}}    |")
print(f"|    Class :  {class_:<{width}}    |")

print("-" * ((width+10) + 3 + 6)) #bottom bar
